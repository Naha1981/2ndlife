   781→### 4. Service pattern example — src/modules/customers/service.ts
   782→```ts
   783→import { db } from '@/lib/db';
   784→import { customers, customerContacts } from '@/lib/db/schema';
   785→import { eq, and } from 'drizzle-orm';
   786→import { tenantWhere } from '@/modules/tenants/tenant-guard';
   787→import { AppError } from '@/shared/errors/types';
   788→import { z } from 'zod';
   789→
   790→export const listCustomersSchema = z.object({ tenantId: z.string(), limit: z.number().min(1).max(100).default(50), offset: z.number().min(0).default(0) });
   791→export type ListCustomersInput = z.infer<typeof listCustomersSchema>;
   792→
   793→export async function listCustomers(input: ListCustomersInput) {
   794→  if (!db) throw new AppError('DATABASE_NOT_CONFIGURED', 'Database not configured');
   795→  const { tenantId, limit, offset } = listCustomersSchema.parse(input);
   796→  const rows = await db.select().from(customers).where(tenantWhere(customers, tenantId)).limit(limit).offset(offset);
   797→  return rows;
   798→}
   799→
   800→export async function getCustomer(tenantId: string, customerId: string) {
   801→  if (!db) throw new AppError('DATABASE_NOT_CONFIGURED', 'Database not configured');
   802→  const row = await db.select().from(customers).where(and(tenantWhere(customers, tenantId), eq(customers.id, customerId))).limit(1);
   803→  if (!row.length) throw new AppError('NOT_FOUND', 'Customer not found', 404);
   804→  return row[0];
   805→}

   809→```ts
   810→import { NextResponse } from 'next/server';
   811→import { auth } from '@clerk/nextjs/server';
   812→import { getTenantContext } from '@/modules/tenants/service';
   813→import { listCustomers } from '@/modules/customers/service';
   814→import { AppError } from '@/shared/errors/types';
   815→
   816→export const dynamic = 'force-dynamic';
   817→
   818→export async function GET(req: Request) {
   819→  try {
   820→    const { userId } = await auth();
   821→    if (!userId) return NextResponse.json({ error: { code: 'UNAUTHORIZED', message: 'Not authenticated' } }, { status: 401 });
   822→    const tenant = await getTenantContext(userId);
   823→    if (!tenant) return NextResponse.json({ error: { code: 'NOT_FOUND', message: 'No tenant' } }, { status: 404 });
   824→    const url = new URL(req.url);
   825→    const rows = await listCustomers({ tenantId: tenant.id, limit: Number(url.searchParams.get('limit') ?? 50), offset: Number(url.searchParams.get('offset') ?? 0) });
   826→    return NextResponse.json({ data: rows });
   827→  } catch (err) {
   828→    if (err instanceof AppError) return NextResponse.json({ error: { code: err.code, message: err.message } }, { status: err.status });
   829→    console.error('customers GET error', err);
   830→    return NextResponse.json({ error: { code: 'INTERNAL', message: 'Unexpected error' } }, { status: 500 });
   831→  }
   832→}
