          </h1>
          <p className="text-xl text-gray-300 mb-8 max-w-3xl">{config.subhead}</p>

              const Icon = iconMap[s.icon];
              return (
                <Card key={s.title} className="bg-[#052e22] border-[#14604a] text-white">
                  <CardContent className="p-6">

      <section className="py-20 px-6 bg-white">

          <h2 className="text-4xl font-bold mb-12 text-center">How It Works</h2>
          <div className="grid md:grid-cols-5 gap-6">
            {config.howItWorks.map((step, i) => (
              <div key={step.title} className="text-center">
                <div className="w-12 h-12 rounded-full bg-[#e9f6ee] text-[#16a34a] flex items-center justify-center mx-auto mb-4 font-bold text-xl">{i+1}</div>
                <h3 className="font-bold mb-2">{step.title}</h3>
                <p className="text-sm text-[#5c6b64]">{step.desc}</p>
              </div>
            ))}
          </div>
        </div>
      </section>

      <section className="bg-[#052e22] text-white py-20 px-6 text-center">

        <p className="text-gray-300 mb-8">{config.ctaSub}</p>
        <Button size="lg" className="bg-[#16a34a] hover:bg-[#15803d]">Book Your Free Demo</Button>
      </section>
    </main>
  );
}

import { eq } from 'drizzle-orm';
export function tenantWhere<T extends { tenantId: any }>(table: T, tenantId: string) {
  if (!tenantId) throw new Error('TENANT_ISOLATION: tenantId required');
  return eq(table.tenantId, tenantId);
}

### 4. Service pattern example — src/modules/customers/service.ts
```ts
import { db } from '@/lib/db';
import { customers, customerContacts } from '@/lib/db/schema';
import { eq, and } from 'drizzle-orm';
import { tenantWhere } from '@/modules/tenants/tenant-guard';
import { AppError } from '@/shared/errors/types';
import { z } from 'zod';

export const listCustomersSchema = z.object({ tenantId: z.string(), limit: z.number().min(1).max(100).default(50), offset: z.number().min(0).default(0) });
export type ListCustomersInput = z.infer<typeof listCustomersSchema>;

export async function listCustomers(input: ListCustomersInput) {
  if (!db) throw new AppError('DATABASE_NOT_CONFIGURED', 'Database not configured');
  const { tenantId, limit, offset } = listCustomersSchema.parse(input);
  const rows = await db.select().from(customers).where(tenantWhere(customers, tenantId)).limit(limit).offset(offset);
  return rows;
}

export async function getCustomer(tenantId: string, customerId: string) {
  if (!db) throw new AppError('DATABASE_NOT_CONFIGURED', 'Database not configured');
  const row = await db.select().from(customers).where(and(tenantWhere(customers, tenantId), eq(customers.id, customerId))).limit(1);
  if (!row.length) throw new AppError('NOT_FOUND', 'Customer not found', 404);
  return row[0];
}
```

### 5. Route handler pattern — src/app/api/v1/customers/route.ts
```ts
import { NextResponse } from 'next/server';
import { auth } from '@clerk/nextjs/server';
import { getTenantContext } from '@/modules/tenants/service';
import { listCustomers } from '@/modules/customers/service';
import { AppError } from '@/shared/errors/types';

export const dynamic = 'force-dynamic';

export async function GET(req: Request) {
  try {
    const { userId } = await auth();
    if (!userId) return NextResponse.json({ error: { code: 'UNAUTHORIZED', message: 'Not authenticated' } }, { status: 401 });
    const tenant = await getTenantContext(userId);
    if (!tenant) return NextResponse.json({ error: { code: 'NOT_FOUND', message: 'No tenant' } }, { status: 404 });
    const url = new URL(req.url);
    const rows = await listCustomers({ tenantId: tenant.id, limit: Number(url.searchParams.get('limit') ?? 50), offset: Number(url.searchParams.get('offset') ?? 0) });
    return NextResponse.json({ data: rows });
  } catch (err) {
    if (err instanceof AppError) return NextResponse.json({ error: { code: err.code, message: err.message } }, { status: err.status });
    console.error('customers GET error', err);
    return NextResponse.json({ error: { code: 'INTERNAL', message: 'Unexpected error' } }, { status: 500 });
  }
}
```
